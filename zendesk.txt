# ── Display driver ─────────────────────────────────────────────
def cmd(c):
    GPIO.output(CS, GPIO.LOW)
    GPIO.output(DC, GPIO.LOW)
    spi.writebytes([c])
    GPIO.output(CS, GPIO.HIGH)

def data(d):
    GPIO.output(CS, GPIO.LOW)
    GPIO.output(DC, GPIO.HIGH)
    spi.writebytes([d])
    GPIO.output(CS, GPIO.HIGH)

def init_display():
    GPIO.output(RST, GPIO.LOW);  time.sleep(0.1)
    GPIO.output(RST, GPIO.HIGH); time.sleep(0.2)
    cmd(0x01); time.sleep(0.2)
    cmd(0x11); time.sleep(0.12)
    cmd(0x3A); data(0x55)
    cmd(0x36); data(0xC8)
    cmd(0x29)

def show_image(img):
    img = img.convert("RGB")
    pixels = list(img.getdata())
    cmd(0x2A)
    data(0x00); data(0x00); data(0x00); data(0xEF)
    cmd(0x2B)
    data(0x00); data(0x00); data(0x01); data(0x3F)
    cmd(0x2C)
    GPIO.output(CS, GPIO.LOW)
    GPIO.output(DC, GPIO.HIGH)
    buf = []
    for r, g, b in pixels:
        c565 = ((r & 0xF8) << 8) | ((g & 0xFC) << 3) | (b >> 3)
        buf.append((c565 >> 8) & 0xFF)
        buf.append(c565 & 0xFF)
        if len(buf) >= 4096:
            spi.writebytes2(buf)
            buf = []
    if buf:
        spi.writebytes2(buf)
    GPIO.output(CS, GPIO.HIGH)

# ── Sensor ─────────────────────────────────────────────────────
def init_sensor():
    bus.write_byte_data(addr, 0x09, 0x40)
    time.sleep(0.1)
    bus.write_byte_data(addr, 0x04, 0x00)
    bus.write_byte_data(addr, 0x05, 0x00)
    bus.write_byte_data(addr, 0x06, 0x00)
    bus.write_byte_data(addr, 0x08, 0x4F)
    bus.write_byte_data(addr, 0x09, 0x03)
    bus.write_byte_data(addr, 0x0A, 0x27)
    bus.write_byte_data(addr, 0x0C, 0xFF)
    bus.write_byte_data(addr, 0x0D, 0xFF)
    time.sleep(0.1)
    print("Sensor initialised")

def flush_fifo():
    bus.write_byte_data(addr, 0x04, 0x00)
    bus.write_byte_data(addr, 0x05, 0x00)
    bus.write_byte_data(addr, 0x06, 0x00)

def read_one_sample():
    wr_ptr = bus.read_byte_data(addr, 0x04) & 0x1F
    rd_ptr = bus.read_byte_data(addr, 0x06) & 0x1F
    n = (wr_ptr - rd_ptr) & 0x1F
    if n == 0:
        return None
    raw = bus.read_i2c_block_data(addr, 0x07, 6)
    red = (raw[0] << 16 | raw[1] << 8 | raw[2]) & 0x3FFFF
    ir  = (raw[3] << 16 | raw[4] << 8 | raw[5]) & 0x3FFFF
    return red, ir

def check_finger():
    """Check multiple samples to confirm finger presence."""
    for _ in range(8):
        s = read_one_sample()
        if s is not None and s[1] > 50000:
            return True
        time.sleep(0.01)
    return False

def collect_samples(n_samples=100, update_cb=None):
    flush_fifo()
    time.sleep(0.05)
    reds, irs = [], []
    for i in range(n_samples):
        t0 = time.time()
        sample = None
        deadline = t0 + 0.08
        while time.time() < deadline:
            sample = read_one_sample()
            if sample is not None:
                break
            time.sleep(0.005)
        if sample is None:
            continue
        red, ir = sample
        if ir < 30000:
            print(f"Finger removed at sample {i}")
            return None, None
        reds.append(red)
        irs.append(ir)
        if update_cb:
            update_cb(reds, irs, i / n_samples)
        elapsed = time.time() - t0
        time.sleep(max(0, 0.04 - elapsed))
    return reds, irs

# ── Signal processing ──────────────────────────────────────────
def bandpass_simple(signal, window=5):
    if len(signal) < window * 2:
        return signal
    dc = []
    for i in range(len(signal)):
        lo = max(0, i - window)
        hi = min(len(signal), i + window + 1)
        dc.append(sum(signal[lo:hi]) / (hi - lo))
    return [signal[i] - dc[i] for i in range(len(signal))]

def detect_peaks(signal, min_distance=10, threshold_ratio=0.4):
    if len(signal) < 4:
        return []
    mn, mx = min(signal), max(signal)
    threshold = mn + (mx - mn) * threshold_ratio
    peaks = []
    last_peak = -min_distance
    for i in range(1, len(signal) - 1):
        if (signal[i] > threshold and
            signal[i] > signal[i-1] and
            signal[i] > signal[i+1] and
            i - last_peak >= min_distance):
            peaks.append(i)
            last_peak = i
    return peaks

def calculate_hr(ir_signal, sample_rate=25):
    if len(ir_signal) < 20:
        return 72
    ac = bandpass_simple(ir_signal, window=6)
    peaks = detect_peaks(ac, min_distance=10, threshold_ratio=0.35)
    if len(peaks) < 2:
        peaks = detect_peaks(ac, min_distance=8, threshold_ratio=0.2)
    if len(peaks) >= 2:
        intervals = [peaks[i+1] - peaks[i] for i in range(len(peaks)-1)]
        avg_interval = sum(intervals) / len(intervals)
        hr = int(60 / (avg_interval / sample_rate))
        return max(45, min(150, hr))
    return 72

def calculate_spo2(red_signal, ir_signal):
    if len(red_signal) < 10 or len(ir_signal) < 10:
        return 98
    def ac_dc(sig):
        mn, mx = min(sig), max(sig)
        dc = sum(sig) / len(sig)
        ac = (mx - mn) / 2.0
        return ac, dc
    ac_r, dc_r   = ac_dc(red_signal)
    ac_ir, dc_ir = ac_dc(ir_signal)
    if dc_r < 1 or dc_ir < 1 or ac_ir < 1:
        return 98
    R = (ac_r / dc_r) / (ac_ir / dc_ir)
    return max(85, min(100, int(110 - 25 * R)))

# ── Drawing helpers ────────────────────────────────────────────
def fill_gradient(draw, y1, y2, color_top, color_bot):
    for y in range(y1, y2):
        t = (y - y1) / max(1, y2 - y1)
        r = int(color_top[0] + (color_bot[0]-color_top[0]) * t)
        g = int(color_top[1] + (color_bot[1]-color_top[1]) * t)
        b = int(color_top[2] + (color_bot[2]-color_top[2]) * t)
        draw.line([(0,y),(WIDTH,y)], fill=(r,g,b))

def rrect(draw, x1, y1, x2, y2, r=10, fill=None, outline=None, width=1):
    draw.rounded_rectangle([(x1,y1),(x2,y2)], radius=r,
                            fill=fill, outline=outline, width=width)

def glow_circle(draw, cx, cy, r, color, layers=3):
    for i in range(layers, 0, -1):
        cr = min(255, color[0]+30)
        cg = min(255, color[1]+30)
        cb = min(255, color[2]+30)
        draw.ellipse([cx-r-i*3, cy-r-i*3, cx+r+i*3, cy+r+i*3],
                     outline=(cr, cg, cb), width=1)
    draw.ellipse([cx-r, cy-r, cx+r, cy+r], fill=color)

def draw_sparkle(draw, x, y, size, color):
    draw.polygon([(x,y-size),(x+size//3,y),(x,y+size),(x-size//3,y)], fill=color)
    draw.polygon([(x-size,y),(x,y-size//3),(x+size,y),(x,y+size//3)], fill=color)

def draw_pulse_wave(draw, ir_signal, x0, y0, w, h, color=MINT):
    rrect(draw, x0-2, y0-2, x0+w+2, y0+h+2, 6, fill=BG_DARK, outline=PURPLE_MID, width=1)
    if len(ir_signal) < 4:
        draw.line([(x0, y0+h//2),(x0+w, y0+h//2)], fill=PURPLE_MID, width=1)
        return
    ac = bandpass_simple(ir_signal, window=8)
    display = ac[-60:] if len(ac) > 60 else ac
    mn, mx = min(display), max(display)
    rng = mx - mn
    if rng < 10:
        draw.line([(x0, y0+h//2),(x0+w, y0+h//2)], fill=PURPLE_MID, width=1)
        draw.text((x0+w//2, y0+h//2-8), "Keep still...",
                  fill=PURPLE_LIGHT, font=font(9), anchor="mm")
        return
    n = len(display)
    pts = []
    for i, v in enumerate(display):
        px = x0 + int(i * w / max(1, n-1))
        py = y0 + h - int((v - mn) / rng * (h - 4)) - 2
        py = max(y0+1, min(y0+h-1, py))
        pts.append((px, py))
    for i in range(len(pts)-1):
        draw.line([pts[i], pts[i+1]], fill=(color[0]//3, color[1]//3, color[2]//3), width=4)
    for i in range(len(pts)-1):
        draw.line([pts[i], pts[i+1]], fill=color, width=2)
    peaks = detect_peaks(display, min_distance=8, threshold_ratio=0.35)
    for p in peaks:
        if 0 < p < n:
            draw.ellipse([pts[p][0]-3, pts[p][1]-3,
                          pts[p][0]+3, pts[p][1]+3], fill=PINK_HOT)

# MAIN LOOP

init_display()
init_sensor()
print("Zen Desk started!")

frame = 0

try:
    while True:
        # ── SCREEN 2: Wait for finger ──────────────────────────
        print("Waiting for finger...")
        frame = 0
        while True:
            # Show place-finger animation
            show_image(screen_place_finger(frame))
            frame = (frame + 1) % 200

            # Check for finger multiple times per frame
            found = False
            for _ in range(5):
                s = read_one_sample()
                if s is not None and s[1] > 50000:
                    found = True
                    break
                time.sleep(0.01)

            if found:
                print("Finger detected!")
                break

            time.sleep(0.05)

        # ── Warm-up: discard first 25 samples ─────────────────
        print("Warming up sensor...")
        flush_fifo()
        for _ in range(25):
            read_one_sample()
            time.sleep(0.04)

        # ── Collect 100 samples at 25Hz (~4 sec) ──────────────
        print("Collecting samples...")
        all_reds, all_irs = [], []
        finger_lost = False

        for sec in range(10, 0, -1):
            for sub in range(25):  # 25 samples per second
                t0 = time.time()
                s = None
                wait = 0
                while s is None and wait < 15:
                    s = read_one_sample()
                    wait += 1
                    time.sleep(0.004)

                if s is None:
                    continue

                red, ir = s
                if ir < 30000:
                    print("Finger lost!")
                    finger_lost = True
                    break

                all_reds.append(red)
                all_irs.append(ir)

                if sub % 5 == 0:
                    show_image(screen_measuring(sec, all_reds, all_irs, frame))
                    frame += 1

                elapsed = time.time() - t0
                time.sleep(max(0, 0.04 - elapsed))

            if finger_lost:
                break

        if finger_lost:
            continue

        if len(all_irs) < 20:
            print(f"Not enough data: {len(all_irs)} samples")
            continue

        # ── Calculate ──────────────────────────────────────────
        hr   = calculate_hr(all_irs, sample_rate=25)
        spo2 = calculate_spo2(all_reds, all_irs)
        mood = get_mood(hr)
        print(f"Result: HR={hr}  SpO2={spo2}  Mood={mood}")

        # ── SCREEN 4: Vitals (10 sec) ──────────────────────────
        t = time.time()
        blink = False
        while time.time() - t < 10:
            show_image(screen_vitals(hr, spo2, mood, all_irs, blink, frame))
            blink = not blink
            frame += 1
            time.sleep(0.6)

        # ── SCREEN 5: Mood (8 sec) ─────────────────────────────
        t = time.time()
        blink = False
        frame = 0
        while time.time() - t < 8:
            show_image(screen_mood(hr, spo2, mood, blink, frame))
            blink = not blink
            frame += 1
            time.sleep(0.5)

        # ── SCREEN 6: Tip (5 sec) ──────────────────────────────
        t = time.time()
        while time.time() - t < 5:
            show_image(screen_tip(mood, hr))
            time.sleep(0.2)

except KeyboardInterrupt:
    GPIO.cleanup()
    print("Zen Desk stopped.")
